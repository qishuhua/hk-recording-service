cmake_minimum_required(VERSION 3.16)
project(hk-recording-service VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HKMEDIA_CORE_TESTS_ONLY "Build dependency-free unit tests only" OFF)
option(HKMEDIA_BUILD_TESTS "Build tests" ON)
if(HKMEDIA_BUILD_TESTS OR HKMEDIA_CORE_TESTS_ONLY)
  enable_testing()
  add_executable(core-test tests/core_test.cpp)
  target_include_directories(core-test PRIVATE src)
  add_test(NAME core-test COMMAND core-test)
endif()
if(HKMEDIA_CORE_TESTS_ONLY)
  return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "Requires Linux x86_64 HCNetSDK")
endif()
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
pkg_check_modules(HTTPLIB QUIET IMPORTED_TARGET cpp-httplib)
if(HTTPLIB_FOUND)
  set(HTTPLIB_TARGET PkgConfig::HTTPLIB)
else()
  find_library(HTTPLIB_TARGET NAMES cpp-httplib REQUIRED)
endif()
set(HCNETSDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/HCNetSDK" CACHE PATH "HCNetSDK root")
if(NOT EXISTS "${HCNETSDK_DIR}/include/HCNetSDK.h" OR NOT EXISTS "${HCNETSDK_DIR}/lib/libhcnetsdk.so")
  message(FATAL_ERROR "Place HCNetSDK in third_party/HCNetSDK or run scripts/import_sdk.sh /path/to/HCNetSDK")
endif()
add_library(hcnetsdk SHARED IMPORTED)
set_target_properties(hcnetsdk PROPERTIES IMPORTED_LOCATION "${HCNETSDK_DIR}/lib/libhcnetsdk.so" IMPORTED_NO_SONAME TRUE INTERFACE_INCLUDE_DIRECTORIES "${HCNETSDK_DIR}/include")
add_executable(hk-recording-service src/main.cpp src/http/http_api.cpp src/recording/recording_download.cpp src/media/mp4_muxer.cpp)
target_include_directories(hk-recording-service PRIVATE src)
target_compile_options(hk-recording-service PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(hk-recording-service PRIVATE PkgConfig::FFMPEG ${HTTPLIB_TARGET} nlohmann_json::nlohmann_json spdlog::spdlog hcnetsdk Threads::Threads)
target_link_options(hk-recording-service PRIVATE "LINKER:--disable-new-dtags")
set_target_properties(hk-recording-service PROPERTIES BUILD_RPATH "${HCNETSDK_DIR}/lib;${HCNETSDK_DIR}/lib/HCNetSDKCom" INSTALL_RPATH "$ORIGIN/../lib;$ORIGIN/../lib/HCNetSDKCom")
install(TARGETS hk-recording-service RUNTIME DESTINATION bin)
